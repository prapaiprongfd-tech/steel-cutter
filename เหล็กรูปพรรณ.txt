    }

    function updateRowCalc(id) {
      const tr = document.getElementById(`row-${id}`);
      if (!tr) return;
      const set = parseFloat(tr.querySelector('.row-set').value) || 0;
      const qty = parseFloat(tr.querySelector('.row-qty').value) || 0;
      tr.querySelector('.row-total').innerText = set * qty;
    }

    function deleteRow(id) {
      const tr = document.getElementById(`row-${id}`);
      if (tr) tr.remove();
    }

    function cloneRow(id) {
      const tr = document.getElementById(`row-${id}`);
      if (!tr) return;
      addNewRow({
        comp: tr.querySelector('.row-comp').value + " (สำเนา)",
        profile: tr.querySelector('.row-profile').value,
        stock: tr.querySelector('.row-stock').value,
        set: tr.querySelector('.row-set').value,
        qty: tr.querySelector('.row-qty').value,
        cutLen: tr.querySelector('.row-cutlen').value,
        miter: tr.querySelector('.row-miter').value
      });
    }

    function moveRow(id, dir) {
      const tr = document.getElementById(`row-${id}`);
      if (!tr) return;
      if (dir === -1 && tr.previousElementSibling) {
        tr.parentNode.insertBefore(tr, tr.previousElementSibling);
      } else if (dir === 1 && tr.nextElementSibling) {
        tr.parentNode.insertBefore(tr.nextElementSibling, tr);
      }
    }

      document.querySelectorAll('.row-stock').forEach(sel => sel.value = type);

    // ฟังก์ชันคำนวณการจัดตัดเหล็กรูปพรรณ (Best-Fit Decreasing)
    function calculateCutting() {
      const kerfM = (parseFloat(document.getElementById('kerfInput').value) || 3) / 1000;
      const rows = document.querySelectorAll('#barcutTbody tr');

      // จัดกลุ่มตามประเภทหน้าตัดเหล็ก
      let sections = {};

      rows.forEach(r => {
        const id = r.id;
        const comp = r.querySelector('.row-comp').value;
        const profile = r.querySelector('.row-profile').value;
        const stockMode = r.querySelector('.row-stock').value;
        const set = parseFloat(r.querySelector('.row-set').value) || 0;
        const qty = parseFloat(r.querySelector('.row-qty').value) || 0;
        const total = set * qty;
        const cutLen = parseFloat(r.querySelector('.row-cutlen').value) || 0;
        const miter = r.querySelector('.row-miter').value;

        if (!sections[profile]) sections[profile] = [];

        sections[profile].push({
          rowId: id, comp, total, cutLen, stockMode, miter
        });
      });

      let boqSummary = {};
      let diagramHtml = '';

      // วนลูปคำนวณทีละหน้าตัด
      for (const [profile, items] of Object.entries(sections)) {
        const profileInfo = STEEL_PROFILES.find(p => p.name === profile) || { weight: 5.0 };
        
        let cuts = [];
        items.forEach(it => {
          for (let i = 0; i < it.total; i++) {
            cuts.push({ comp: it.comp, length: it.cutLen, rowId: it.rowId });
          }
        });

        // เรียงจากยาวไปสั้น
        cuts.sort((a, b) => b.length - a.length);

        let bars = [];
        const stdLen = 6.00; // ค่ามาตรฐาน 6 เมตร

        cuts.forEach(c => {
          let bestBar = null;
          let minWaste = Infinity;

          for (let b of bars) {
            let spaceNeeded = c.length + (b.cuts.length > 0 ? kerfM : 0);
            if (b.remaining >= spaceNeeded) {
              let waste = b.remaining - spaceNeeded;
              if (waste < minWaste) {
                minWaste = waste;
                bestBar = b;
              }
            }
          }

          if (!bestBar) {
            bestBar = {
              id: `ท่อนที่ ${bars.length + 1}`,
              originalLen: stdLen,
              remaining: stdLen,
              cuts: []
            };
            bars.push(bestBar);
          }

          let spaceNeeded = c.length + (bestBar.cuts.length > 0 ? kerfM : 0);
          bestBar.remaining -= spaceNeeded;
          bestBar.cuts.push(c);
        });

        // บันทึกผลลงตาราง Barcut
        items.forEach(it => {
          const tr = document.getElementById(it.rowId);
          if (tr) {
            tr.querySelector('.row-bars').innerText = `${bars.length} ท่อน`;
            tr.querySelector('.row-reuse').innerText = `ดึงใช้ร่วมในกลุ่ม ${profile.split(' ')[0]}`;
            
            // คำนวณเศษเฉลี่ย
            let totalRem = bars.reduce((s, b) => s + b.remaining, 0);
            let scrapKg = (totalRem * profileInfo.weight).toFixed(1);
            
            tr.querySelector('.row-offcut').innerText = `เหลือปลายรวม ${totalRem.toFixed(2)} ม.`;
            tr.querySelector('.row-scrapg').innerText = scrapKg;
          }
        });

        // รวมยอดเพื่อสร้างตาราง BOQ
        let totalUsedLen = cuts.reduce((s, c) => s + c.length, 0);
        let totalPurchasedLen = bars.length * stdLen;
        let totalPurchasedKg = totalPurchasedLen * profileInfo.weight;
        let totalScrapLen = bars.reduce((s, b) => s + b.remaining, 0);
        let yieldPct = totalPurchasedLen > 0 ? (totalUsedLen / totalPurchasedLen * 100).toFixed(1) : 0;

        boqSummary[profile] = {
          unitWeight: profileInfo.weight,
          usedLen: totalUsedLen.toFixed(2),
          barCount: bars.length,
          scrapLen: totalScrapLen.toFixed(2),
          purchasedKg: totalPurchasedKg.toFixed(1),
          yieldPct: yieldPct
        };

        // สร้าง Visual แผนผังการตัด
        diagramHtml += `
          <div class="border rounded-lg p-3 bg-slate-50 space-y-2">
            <div class="flex justify-between items-center font-bold text-slate-800 text-xs border-b pb-1">
              <span>🔩 ${profile} (ใช้เหล็กใหม่ทั้งหมด: ${bars.length} เส้น)</span>
              <span class="text-emerald-700">Yield รวม: ${yieldPct}%</span>
            </div>
            <div class="space-y-2 pt-1">
        `;

        bars.forEach(b => {
          let cutsVisual = '';
          b.cuts.forEach(c => {
            let pct = (c.length / b.originalLen * 100).toFixed(1);
            cutsVisual += `
              <div style="width: ${pct}%" class="bg-blue-600 text-white text-[10px] flex items-center justify-center border-r border-white/40 truncate py-1" title="${c.comp}: ${c.length} ม.">
                ${c.comp} (${c.length}ม.)
              </div>
            `;
          });
          let remPct = (b.remaining / b.originalLen * 100).toFixed(1);
          let isKeep = b.remaining >= 0.5; // เศษยาวกว่า 50 ซม. เก็บเข้าคลัง
          cutsVisual += `
            <div style="width: ${remPct}%" class="${isKeep ? 'bg-purple-200 text-purple-900 font-bold' : 'bg-slate-300 text-slate-600'} text-[9px] flex items-center justify-center truncate py-1">
              ${isKeep ? `สต็อก +${b.remaining.toFixed(2)}ม.` : `เศษ ${b.remaining.toFixed(2)}ม.`}
            </div>
          `;

          diagramHtml += `
            <div class="flex items-center gap-2">
              <span class="w-14 font-mono text-[10px] text-slate-500">${b.id}</span>
              <div class="flex-1 bg-slate-200 rounded flex overflow-hidden shadow-inner h-6">
                ${cutsVisual}
              </div>
            </div>
          `;
        });

        diagramHtml += `</div></div>`;
      }

      // แสดงข้อมูลในหน้าต่าง BOQ
      let boqRows = '';
      for (const [pName, b] of Object.entries(boqSummary)) {
        boqRows += `
          <tr class="border-b hover:bg-slate-50">
            <td class="p-2 font-bold text-slate-800 border">${pName}</td>
            <td class="p-2 text-right border font-mono">${b.unitWeight.toFixed(2)}</td>
            <td class="p-2 text-right border font-mono">${b.usedLen} ม.</td>
            <td class="p-2 text-center border font-mono font-bold text-blue-700">${b.barCount} เส้น</td>
            <td class="p-2 text-center border text-slate-400">-</td>
            <td class="p-2 text-right border font-mono text-purple-700 font-semibold">${b.scrapLen} ม.</td>
            <td class="p-2 text-right border font-mono font-bold text-emerald-800">${b.purchasedKg} กก.</td>
            <td class="p-2 text-right border font-mono font-bold ${b.yieldPct >= 85 ? 'text-emerald-600' : 'text-amber-600'}">${b.yieldPct}%</td>
          </tr>
        `;
      }
      document.getElementById('boqTbody').innerHTML = boqRows || '<tr><td colspan="8" class="text-center p-4">ไม่มีรายการ</td></tr>';
      document.getElementById('cuttingDiagramContainer').innerHTML = diagramHtml || '<p class="text-center p-4 text-slate-400">ไม่มีข้อมูล</p>';
    }
